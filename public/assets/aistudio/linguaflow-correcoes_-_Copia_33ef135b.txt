+    private fun rms(buf: ByteArray, bytes: Int): Double {
+        val samples = bytes / 2
+        if (samples == 0) return 0.0
+        var sum = 0.0
+        var i = 0
+        while (i + 1 < bytes) {
+            val sample = ((buf[i + 1].toInt() shl 8) or (buf[i].toInt() and 0xFF)).toShort().toInt()
+            sum += sample.toDouble() * sample
+            i += 2
+        }
